/**
 * ==============================================================================
 * 🚀 가족 가계부 3개 시트 (월별관리 · 데이터 · 기타정리자료) 실시간 양방향 자동화 스크립트
 * 문서 ID: 14zMkg5XNMw1H1nF-_ZxM0s7pu-IEf2QPdwZUNhpUYF0
 * ==============================================================================
 */

function onOpen() {
  try {
    var ui = SpreadsheetApp.getUi();
    ui.createMenu('🚀 가계부 0초 자동화')
      .addItem('⚡ 0초 웹앱 강제 동기화', 'syncWebData')
      .addItem('📊 월별 지출 합계 자동 재계산', 'recalculateMonthlyTotals')
      .addItem('🗑️ 데이터 시트 빈 행 정리', 'cleanEmptyRows')
      .addToUi();
  } catch (e) {}
}

function onEdit(e) {
  try {
    recalculateMonthlyTotals();
  } catch (err) {}
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var monthlySheet = ss.getSheetByName("월별관리") ? ss.getSheetByName("월별관리").getDataRange().getDisplayValues() : [];
  var dataSheet = ss.getSheetByName("데이터") ? ss.getSheetByName("데이터").getDataRange().getDisplayValues() : [];
  var miscSheet = ss.getSheetByName("기타정리자료") ? ss.getSheetByName("기타정리자료").getDataRange().getDisplayValues() : [];
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    timestamp: new Date().toISOString(),
    monthly: monthlySheet,
    data: dataSheet,
    misc: miscSheet
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("데이터") || ss.getSheets()[0];
    var data = sheet.getDataRange().getValues();

    if (action === "delete") {
      var targetDesc = payload.desc;
      for (var i = data.length - 1; i >= 1; i--) {
        if ((targetDesc && data[i][4] == targetDesc) || (payload.rowIndex && (i + 1) == payload.rowIndex)) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      recalculateMonthlyTotals();
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "삭제 완료" })).setMimeType(ContentService.MimeType.JSON);
    }
    else if (action === "add") {
      sheet.appendRow(payload.row);
      recalculateMonthlyTotals();
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "추가 완료" })).setMimeType(ContentService.MimeType.JSON);
    }
    else if (action === "edit") {
      var targetDesc = payload.desc;
      for (var i = 1; i < data.length; i++) {
        if ((i + 1) == payload.rowIndex || data[i][4] == targetDesc) {
          sheet.getRange(i + 1, 1, 1, payload.row.length).setValues([payload.row]);
          break;
        }
      }
      recalculateMonthlyTotals();
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "수정 완료" })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function recalculateMonthlyTotals() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dataSheet = ss.getSheetByName("데이터");
  var monthlySheet = ss.getSheetByName("월별관리");
  if (!dataSheet || !monthlySheet) return;

  var dataValues = dataSheet.getDataRange().getValues();
  var junSum = 0;
  var jihSum = 0;

  for (var i = 1; i < dataValues.length; i++) {
    var name = dataValues[i][2];
    var amountStr = String(dataValues[i][5] || '0');
    var val = parseInt(amountStr.replace(/[^0-9]/g, '')) || 0;
    if (name === "준영") junSum += val;
    if (name === "지헌") jihSum += val;
  }
}

function cleanEmptyRows() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("데이터");
  if (!sheet) return;
  var data = sheet.getDataRange().getValues();
  for (var i = data.length - 1; i >= 1; i--) {
    if (!data[i][1] && !data[i][2] && !data[i][4]) {
      sheet.deleteRow(i + 1);
    }
  }
}
